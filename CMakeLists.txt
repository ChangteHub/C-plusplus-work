cmake_minimum_required(VERSION 3.16)
project(SchoolPersonnelManagement LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(personnel_models person.cpp teacher.cpp adminstaff.cpp student.cpp worker.cpp graduate.cpp)
target_include_directories(personnel_models PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(school_sqlite STATIC backend/sqlite3.c)
target_include_directories(school_sqlite PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/backend)
target_link_libraries(school_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(school_sqlite PUBLIC m)
endif()
add_executable(server backend/web_server.cpp backend/service.cpp backend/dao.cpp backend/person_codec.cpp backend/academy.cpp backend/crypto.cpp backend/routes.cpp)
target_link_libraries(server PRIVATE personnel_models school_sqlite Threads::Threads)
if(WIN32)
    target_link_libraries(server PRIVATE ws2_32 bcrypt)
else()
    find_package(OpenSSL REQUIRED)
    target_link_libraries(server PRIVATE OpenSSL::Crypto)
endif()
add_executable(school_system main.cpp personmanager.cpp)
target_link_libraries(school_system PRIVATE personnel_models)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_models tests/test_models.cpp)
    target_link_libraries(test_models PRIVATE personnel_models)
    if(MSVC)
        target_compile_options(test_models PRIVATE /UNDEBUG)
    else()
        target_compile_options(test_models PRIVATE -UNDEBUG)
    endif()
    add_test(NAME personnel_model_invariants COMMAND test_models)
endif()
if(MSVC)
    target_compile_options(personnel_models PRIVATE /utf-8 /W4)
    target_compile_options(server PRIVATE /utf-8 /W4)
    target_compile_options(school_system PRIVATE /utf-8 /W4)
else()
    target_compile_options(personnel_models PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(server PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(school_system PRIVATE -Wall -Wextra -Wpedantic)
endif()
